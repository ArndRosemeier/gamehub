GameHub

A small, in-memory relay for TypeScript multiplayer games. No per-game server code: a game creates a room, one player hosts the simulation, everyone else syncs through it.

Connect a game

import { connect } from "https://games.futuremagic.de/sdk.js";

const hub = connect({ app: "arena", appKey: "ghk_…", name: "you" });
await hub.ready;

hub.watchLobby((rooms) => render(rooms));
await hub.create({ name: "Duel", maxPlayers: 4 });

hub.on("state", (world) => { if (!hub.isHost) local = world; });
hub.on("delta", (d)     => { if (!hub.isHost) apply(local, d); });
if (hub.isHost) { hub.setState(local); hub.sendDelta(step(local)); }
else hub.send({ type: "input", input: controls }, hub.host);
